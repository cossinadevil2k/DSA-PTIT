1:CẤU HÌNH SW_INSIDE
enable
configure terminal

!--- 1. Cấu hình cơ bản & Bảo mật
hostname SW0-INSIDE
enable secret Cisco123!

line console 0
 password Cisco123!
 login
 logging synchronous
exit

!--- 2. Cấu hình SSH
ip domain-name Ptit.com
crypto key generate rsa general-keys modulus 1024
username admin privilege 15 secret Admin123@

line vty 0 15
 login local
 transport input ssh
exit

!--- 3. Định tuyến Layer 3 & Tạo VLAN
ip routing

vlan 10
 name LAN_PC0
vlan 20
 name LAN_PC1
exit

!--- 4. Gán cổng Access cho Client
interface FastEthernet0/5
 switchport mode access
 switchport access vlan 10

interface FastEthernet0/15
 switchport mode access
 switchport access vlan 20

!--- 5. Cấu hình SVI (Gateway các VLAN)
interface Vlan10
 ip address 192.168.10.1 255.255.255.0
 no shutdown

interface Vlan20
 ip address 192.168.20.1 255.255.255.0
 no shutdown

!--- 6. Cấu hình cổng kết nối lên ASA0 (Gig1/2)
interface GigabitEthernet0/1
 no switchport
 ip address 10.10.10.2 255.255.255.252
 no shutdown

!--- 7. Cấu hình DHCP Server cho 2 VLAN
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp pool POOL_VLAN10
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 10.10.20.2

ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp pool POOL_VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 10.10.20.2

!--- 8. Default Route trỏ lên ASA
ip route 0.0.0.0 0.0.0.0 10.10.10.1
exit

write memory


2. CẤU HÌNH ASA

enable
configure terminal


hostname ASA0
enable password Cisco123!
passwd Cisco123!


interface GigabitEthernet1/2
 nameif inside
 security-level 100
 ip address 10.10.10.1 255.255.255.252
 no shutdown

interface GigabitEthernet1/3
 nameif dmz
 security-level 50
 ip address 10.10.20.1 255.255.255.0
 no shutdown

interface GigabitEthernet1/1
 nameif outside
 security-level 0
 ip address 10.0.0.1 255.255.255.252
 no shutdown

!--- 3. Cấu hình Định tuyến Tĩnh (Dùng 'route', không dùng 'ip route')
route outside 0.0.0.0 0.0.0.0 10.0.0.2
route inside 192.168.10.0 255.255.255.0 10.10.10.2
route inside 192.168.20.0 255.255.255.0 10.10.10.2

!--- 4. Dynamic NAT cho dải Inside ra Outside & DMZ
object network INSIDE_VLAN10
 subnet 192.168.10.0 255.255.255.0
 nat (inside,outside) dynamic interface
 nat (inside,dmz) dynamic interface

object network INSIDE_VLAN20
 subnet 192.168.20.0 255.255.255.0
 nat (inside,outside) dynamic interface
 nat (inside,dmz) dynamic interface

!--- 5. Access Control List mở cổng cho Outside vào DMZ (Web/DNS/Ping)
access-list OUTSIDE_TO_DMZ extended permit tcp any host 10.10.20.3 eq www
access-list OUTSIDE_TO_DMZ extended permit udp any host 10.10.20.2 eq domain
access-list OUTSIDE_TO_DMZ extended permit icmp any any
access-group OUTSIDE_TO_DMZ in interface outside

!--- 6. Bật Inspect ICMP & DNS
policy-map global_policy
 class inspection_default
  inspect icmp
  inspect dns

!--- 7. Cấu hình SSH chuẩn ASA Packet Tracer
domain-name Ptit.com
crypto key generate rsa modulus 1024
username admin password Admin123@

ssh 192.168.10.0 255.255.255.0 inside
ssh 192.168.20.0 255.255.255.0 inside
ssh 10.10.10.0 255.255.255.0 inside
ssh timeout 5
exit

write memory

exit
